-- PROPOSED ONLY. Explicit DBA review and approved migration required.
-- NEVER placed in server/drizzle auto-discovery paths.
-- A durable attempt must be reserved before ANY Square API call.
-- No rows in this table are proof of payment or entitlement.
CREATE TABLE IF NOT EXISTS wedding_square_enrollment_attempts (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id varchar NOT NULL REFERENCES users(id),
  square_environment text NOT NULL DEFAULT 'sandbox'
    CHECK (square_environment = 'sandbox'),
  tier text NOT NULL CHECK (tier IN ('premium', 'elite')),
  consent_terms_version text NOT NULL,
  consented_at timestamptz NOT NULL,
  consented_amount_cents integer NOT NULL CHECK (consented_amount_cents IN (2499,4999)),
  consented_currency text NOT NULL CHECK (consented_currency = 'USD'),
  trial_expires_at timestamptz,
  requested_start_date date,
  requested_timezone text NOT NULL DEFAULT 'UTC' CHECK (requested_timezone = 'UTC'),
  square_location_id text NOT NULL,
  square_plan_variation_id text NOT NULL,
  square_customer_id text NOT NULL,
  square_card_id text NOT NULL,
  idempotency_key text NOT NULL UNIQUE CHECK (length(idempotency_key) BETWEEN 1 AND 45),
  request_fingerprint text NOT NULL,
  provider_subscription_id text,
  status text NOT NULL DEFAULT 'reserved'
    CHECK (status IN ('reserved','submitting','needs_reconciliation','confirmed','aborted')),
  last_error_code text,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  CHECK (
    (tier = 'premium' AND consented_amount_cents = 2499)
    OR (tier = 'elite' AND consented_amount_cents = 4999)
  ),
  CHECK (trial_expires_at IS NULL OR
    (requested_start_date IS NOT NULL AND
     (requested_start_date::timestamp AT TIME ZONE 'UTC') >= trial_expires_at)),
  CHECK (provider_subscription_id IS NULL OR status IN ('needs_reconciliation','confirmed'))
);
CREATE INDEX IF NOT EXISTS wedding_square_enrollment_attempts_user_idx
  ON wedding_square_enrollment_attempts(user_id, created_at DESC);
CREATE UNIQUE INDEX IF NOT EXISTS wedding_square_enrollment_attempts_provider_idx
  ON wedding_square_enrollment_attempts(square_environment,provider_subscription_id)
  WHERE provider_subscription_id IS NOT NULL;
-- One unresolved attempt per user. Failed/ambiguous Square calls remain
-- needs_reconciliation, never automatically retried using a new key.
CREATE UNIQUE INDEX IF NOT EXISTS wedding_square_enrollment_attempts_unresolved_user_idx
  ON wedding_square_enrollment_attempts(user_id)
  WHERE status IN ('reserved','submitting','needs_reconciliation','confirmed');
